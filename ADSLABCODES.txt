1) BST TREE
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left;
    struct Node *right;
};

/* Create and insert a node */
struct Node* insert(struct Node* root, int value) {
    if (root == NULL) {
        struct Node* newNode =
            (struct Node*)malloc(sizeof(struct Node));

        newNode->data = value;
        newNode->left = NULL;
        newNode->right = NULL;

        return newNode;
    }

    if (value < root->data)
        root->left = insert(root->left, value);

    else if (value > root->data)
        root->right = insert(root->right, value);

    return root;
}

/* Search for a key */
struct Node* search(struct Node* root, int key) {
    if (root == NULL || root->data == key)
        return root;

    if (key < root->data)
        return search(root->left, key);

    return search(root->right, key);
}

/* Find minimum value node */
struct Node* minValueNode(struct Node* node) {
    struct Node* current = node;

    while (current->left != NULL)
        current = current->left;

    return current;
}

/* Delete a node */
struct Node* deleteNode(struct Node* root, int key) {
    if (root == NULL)
        return root;

    if (key < root->data)
        root->left = deleteNode(root->left, key);

    else if (key > root->data)
        root->right = deleteNode(root->right, key);

    else {
        /* Node with no child or one child */
        if (root->left == NULL) {
            struct Node* temp = root->right;
            free(root);
            return temp;
        }

        else if (root->right == NULL) {
            struct Node* temp = root->left;
            free(root);
            return temp;
        }

        /* Node with two children */
        struct Node* temp = minValueNode(root->right);

        root->data = temp->data;

        root->right =
            deleteNode(root->right, temp->data);
    }

    return root;
}

/* Inorder traversal */
void inorder(struct Node* root) {
    if (root != NULL) {
        inorder(root->left);
        printf("%d ", root->data);
        inorder(root->right);
    }
}

int main() {
    struct Node* root = NULL;
    int n, value, key;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter elements:\n");
	int i;
    for (i = 0; i < n; i++) {
    	int i;
        scanf("%d", &value);
        root = insert(root, value);
    }

    printf("Inorder traversal: ");

    printf("\nEnter element to search: ");
    scanf("%d", &key);

    if (search(root, key) != NULL)
        printf("Element found\n");
    else
        printf("Element not found\n");

    printf("Enter element to delete: ");
    scanf("%d", &key);

    root = deleteNode(root, key);

    printf("After deletion: ");
    inorder(root);

    return 0;
}

2) Merge Sort
#include <stdio.h>

void merge(int a[], int low, int mid, int high) {
    int i = low;
    int j = mid + 1;
    int k = low;
    int temp[100];

    while (i <= mid && j <= high) {
        if (a[i] < a[j]) {
            temp[k] = a[i];
            i++;
        }
        else {
            temp[k] = a[j];
            j++;
        }
        k++;
    }

    while (i <= mid) {
        temp[k] = a[i];
        i++;
        k++;
    }

    while (j <= high) {
        temp[k] = a[j];
        j++;
        k++;
    }

    for (i = low; i <= high; i++)
        a[i] = temp[i];
}

void mergeSort(int a[], int low, int high) {
    if (low < high) {
        int mid = (low + high) / 2;

        mergeSort(a, low, mid);
        mergeSort(a, mid + 1, high);

        merge(a, low, mid, high);
    }
}

int main() {
    int a[100], n, i;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter elements:\n");

    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    mergeSort(a, 0, n - 1);

    printf("Sorted array: ");

    for (i = 0; i < n; i++)
        printf("%d ", a[i]);

    return 0;
}


3) Heap Sort
#include <stdio.h>

void heapify(int a[], int n, int i) {
    int largest = i;
    int left = 2 * i + 1;
    int right = 2 * i + 2;

    if (left < n && a[left] > a[largest])
        largest = left;

    if (right < n && a[right] > a[largest])
        largest = right;

    if (largest != i) {
        int temp = a[i];
        a[i] = a[largest];
        a[largest] = temp;

        heapify(a, n, largest);
    }
}

void heapSort(int a[], int n) {

    // Build max heap
    int i;
    for (i = n / 2 - 1; i >= 0; i--)
        heapify(a, n, i);

    // Extract elements one by one
    int i;
    for (i = n - 1; i > 0; i--) {
		int i;
        int temp = a[0];
        a[0] = a[i];
        a[i] = temp;

        heapify(a, i, 0);
    }
}

    heapSort(a, n);

4) Quick Sort
#include <stdio.h>

void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

int partition(int a[], int low, int high) {
    int pivot = a[high];
    int i = low - 1;

    for (int j = low; j < high; j++) {

        if (a[j] < pivot) {
            i++;
            swap(&a[i], &a[j]);
        }
    }

    swap(&a[i + 1], &a[high]);

    return i + 1;
}

void quickSort(int a[], int low, int high) {
    if (low < high) {

        int p = partition(a, low, high);

        quickSort(a, low, p - 1);
        quickSort(a, p + 1, high);
    }
}

int main() {
    int a[100], n;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter elements:\n");

    for (int i = 0; i < n; i++)
        scanf("%d", &a[i]);

    quickSort(a, 0, n - 1);

    printf("Sorted array: ");

    for (int i = 0; i < n; i++)
        printf("%d ", a[i]);

    return 0;
}



5) Min Max heap
#include <stdio.h>

#define MAX 100

int heap[MAX];
int n = 0;

/* Find level of a node */
int level(int i) {
    int l = 0;

    while (i > 1) {
        i = i / 2;
        l++;
    }

    return l;
}

/* Check whether node is at Min level */
int isMinLevel(int i) {
    return level(i) % 2 == 0;
}

/* Swap two elements */
void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

/* Insert an element */
void insert(int x) {
    int i = ++n;

    heap[i] = x;

    if (i == 1)
        return;

    int p = i / 2;

    if (isMinLevel(i)) {

        if (heap[i] > heap[p]) {
            swap(&heap[i], &heap[p]);

            i = p;

            while (i > 3) {
                int gp = i / 4;

                if (heap[i] < heap[gp]) {
                    swap(&heap[i], &heap[gp]);
                    i = gp;
                }
                else
                    break;
            }
        }
        else {
            while (i > 3) {
                int gp = i / 4;

                if (heap[i] < heap[gp]) {
                    swap(&heap[i], &heap[gp]);
                    i = gp;
                }
                else
                    break;
            }
        }
    }
    else {

        if (heap[i] < heap[p]) {
            swap(&heap[i], &heap[p]);

            i = p;

            while (i > 3) {
                int gp = i / 4;

                if (heap[i] > heap[gp]) {
                    swap(&heap[i], &heap[gp]);
                    i = gp;
                }
                else
                    break;
            }
        }
        else {
            while (i > 3) {
                int gp = i / 4;

                if (heap[i] > heap[gp]) {
                    swap(&heap[i], &heap[gp]);
                    i = gp;
                }
                else
                    break;
            }
        }
    }
}

/* Display heap */
void display() {
    for (int i = 1; i <= n; i++)
        printf("%d ", heap[i]);

    printf("\n");
}

/* Search an element */
void search(int x) {
    for (int i = 1; i <= n; i++) {
        if (heap[i] == x) {
            printf("Element found at position %d\n", i);
            return;
        }
    }

    printf("Element not found\n");
}

5) B tree
#include <stdio.h>
#include <stdlib.h>

#define MAX 2
#define MIN 1

struct BTreeNode {
    int key[MAX + 1];
    struct BTreeNode *child[MAX + 2];
    int n;
    int leaf;
};

struct BTreeNode *root = NULL;


/* Create a new node */
struct BTreeNode* createNode(int leaf) {
    struct BTreeNode *node;

    node = (struct BTreeNode*)malloc(sizeof(struct BTreeNode));

    node->n = 0;
    node->leaf = leaf;

    for (int i = 0; i < MAX + 2; i++)
        node->child[i] = NULL;

    return node;
}


/* Search */
int search(struct BTreeNode *node, int x) {

    int i = 0;

    while (i < node->n && x > node->key[i])
        i++;

    if (i < node->n && x == node->key[i])
        return 1;

    if (node->leaf)
        return 0;

    return search(node->child[i], x);
}


/* Split a full child */
void splitChild(struct BTreeNode *parent, int pos) {

    struct BTreeNode *full = parent->child[pos];

    struct BTreeNode *newNode = createNode(full->leaf);

    int middle = full->key[MIN];

    newNode->n = MIN;

    newNode->key[0] = full->key[MIN + 1];

    if (!full->leaf) {
        newNode->child[0] = full->child[MIN + 1];
        newNode->child[1] = full->child[MIN + 2];
    }

    full->n = MIN;

    for (int j = parent->n; j >= pos + 1; j--)
        parent->child[j + 1] = parent->child[j];

    parent->child[pos + 1] = newNode;

    for (int j = parent->n - 1; j >= pos; j--)
        parent->key[j + 1] = parent->key[j];

    parent->key[pos] = middle;

    parent->n++;
}


/* Insert into a non-full node */
void insertNonFull(struct BTreeNode *node, int x) {

    int i = node->n - 1;

    if (node->leaf) {

        while (i >= 0 && x < node->key[i]) {
            node->key[i + 1] = node->key[i];
            i--;
        }

        node->key[i + 1] = x;
        node->n++;
    }

    else {

        while (i >= 0 && x < node->key[i])
            i--;

        i++;

        if (node->child[i]->n == MAX) {

            splitChild(node, i);

            if (x > node->key[i])
                i++;
        }

        insertNonFull(node->child[i], x);
    }
}


/* Insert */
void insert(int x) {

    if (root == NULL) {
        root = createNode(1);
        root->key[0] = x;
        root->n = 1;
        return;
    }

    if (root->n == MAX) {

        struct BTreeNode *newRoot = createNode(0);

        newRoot->child[0] = root;

        root = newRoot;

        splitChild(root, 0);

        insertNonFull(root, x);
    }

    else {
        insertNonFull(root, x);
    }
}


/* Find predecessor */
int getPredecessor(struct BTreeNode *node, int index) {

    struct BTreeNode *current = node->child[index];

    while (!current->leaf)
        current = current->child[current->n];

    return current->key[current->n - 1];
}


/* Find successor */
int getSuccessor(struct BTreeNode *node, int index) {

    struct BTreeNode *current = node->child[index + 1];

    while (!current->leaf)
        current = current->child[0];

    return current->key[0];
}


/* Merge two children */
void merge(struct BTreeNode *node, int index) {

    struct BTreeNode *child = node->child[index];
    struct BTreeNode *sibling = node->child[index + 1];

    child->key[MIN] = node->key[index];

    child->key[MIN + 1] = sibling->key[0];

    child->n = 3;

    for (int i = 0; i < sibling->n; i++)
        child->key[i + MIN + 1] = sibling->key[i];

    for (int i = 0; i <= sibling->n; i++)
        child->child[i + MIN + 1] = sibling->child[i];

    for (int i = index + 1; i < node->n; i++)
        node->key[i - 1] = node->key[i];

    for (int i = index + 2; i <= node->n; i++)
        node->child[i - 1] = node->child[i];

    node->n--;

    free(sibling);
}


/* Delete */
void deleteKey(struct BTreeNode *node, int x) {

    int index = 0;

    while (index < node->n && x > node->key[index])
        index++;

    /* Key found */
    if (index < node->n && node->key[index] == x) {

        if (node->leaf) {

            for (int i = index + 1; i < node->n; i++)
                node->key[i - 1] = node->key[i];

            node->n--;
        }

        else {

            if (node->child[index]->n > MIN) {

                int pred = getPredecessor(node, index);

                node->key[index] = pred;

                deleteKey(node->child[index], pred);
            }

            else if (node->child[index + 1]->n > MIN) {

                int succ = getSuccessor(node, index);

                node->key[index] = succ;

                deleteKey(node->child[index + 1], succ);
            }

            else {

                merge(node, index);

                deleteKey(node->child[index], x);
            }
        }
    }

    /* Key not found */
    else {

        if (node->leaf)
            return;

        if (node->child[index]->n == MIN) {

            if (index > 0 && node->child[index - 1]->n > MIN) {

                struct BTreeNode *child = node->child[index];
                struct BTreeNode *left = node->child[index - 1];

                for (int i = child->n; i > 0; i--)
                    child->key[i] = child->key[i - 1];

                child->key[0] = node->key[index - 1];

                node->key[index - 1] = left->key[left->n - 1];

                child->n++;
                left->n--;
            }

            else if (index < node->n &&
                     node->child[index + 1]->n > MIN) {

                struct BTreeNode *child = node->child[index];
                struct BTreeNode *right = node->child[index + 1];

                child->key[child->n] = node->key[index];

                node->key[index] = right->key[0];

                child->n++;
                right->n--;

                for (int i = 0; i < right->n; i++)
                    right->key[i] = right->key[i + 1];
            }

            else {

                if (index < node->n)
                    merge(node, index);
                else
                    merge(node, index - 1);

                if (index > node->n)
                    index--;
            }
        }

        deleteKey(node->child[index], x);
    }
}


/* Display B-Tree */
void display(struct BTreeNode *node) {

    if (node == NULL)
        return;

    for (int i = 0; i < node->n; i++) {

        if (!node->leaf)
            display(node->child[i]);

        printf("%d ", node->key[i]);
    }

    if (!node->leaf)
        display(node->child[node->n]);
}


int main() {

    int choice, x;

    while (1) {

        printf("\n\n1. Insert");
        printf("\n2. Delete");
        printf("\n3. Search");
        printf("\n4. Display");
        printf("\n5. Exit");

        printf("\nEnter choice: ");
        scanf("%d", &choice);

        switch (choice) {

            case 1:
                printf("Enter element: ");
                scanf("%d", &x);

                insert(x);
                break;

            case 2:
                printf("Enter element to delete: ");
                scanf("%d", &x);

                if (root != NULL)
                    deleteKey(root, x);

                if (root != NULL && root->n == 0) {

                    struct BTreeNode *temp = root;

                    if (root->leaf)
                        root = NULL;
                    else
                        root = root->child[0];

                    free(temp);
                }

                break;

            case 3:
                printf("Enter element to search: ");
                scanf("%d", &x);

                if (root != NULL && search(root, x))
                    printf("Element found\n");
                else
                    printf("Element not found\n");

                break;

            case 4:
                printf("B-Tree: ");
                display(root);
                printf("\n");
                break;

            case 5:
                return 0;

            default:
                printf("Invalid choice\n");
        }
    }
}


6) leftist tree
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    int npl;
    struct Node *left;
    struct Node *right;
};

struct Node* createNode(int value) {
    struct Node *newNode;

    newNode = (struct Node*)malloc(sizeof(struct Node));

    newNode->data = value;
    newNode->npl = 0;
    newNode->left = NULL;
    newNode->right = NULL;

    return newNode;
}

/* Merge two Leftist Trees */
struct Node* merge(struct Node *h1, struct Node *h2) {

    if (h1 == NULL)
        return h2;

    if (h2 == NULL)
        return h1;

    /* Smaller root becomes the root */
    if (h2->data < h1->data) {
        struct Node *temp = h1;
        h1 = h2;
        h2 = temp;
    }

    /* Merge with right subtree */
    h1->right = merge(h1->right, h2);

    /* Leftist property */
    if (h1->left == NULL ||
        h1->left->npl < h1->right->npl) {

        struct Node *temp = h1->left;
        h1->left = h1->right;
        h1->right = temp;
    }

    /* Update NPL */
    if (h1->right == NULL)
        h1->npl = 0;
    else
        h1->npl = h1->right->npl + 1;

    return h1;
}


/* Insert */
struct Node* insert(struct Node *root, int value) {

    struct Node *newNode = createNode(value);

    return merge(root, newNode);
}


/* Delete minimum element */
struct Node* deleteMin(struct Node *root) {

    if (root == NULL) {
        printf("Heap is empty\n");
        return NULL;
    }

    struct Node *leftTree = root->left;
    struct Node *rightTree = root->right;

    printf("Deleted element: %d\n", root->data);

    free(root);

    return merge(leftTree, rightTree);
}


/* Search */
int search(struct Node *root, int value) {

    if (root == NULL)
        return 0;

    if (root->data == value)
        return 1;

    if (search(root->left, value))
        return 1;

    return search(root->right, value);
}


/* Inorder display */
void display(struct Node *root) {

    if (root == NULL)
        return;

    display(root->left);
    printf("%d ", root->data);
    display(root->right);
}

    struct Node *root = NULL;

    int choice, value;

    while (1) {

        printf("\n\n1. Insert");
        printf("\n2. Delete Minimum");
        printf("\n3. Search");
        printf("\n4. Display");
        printf("\n5. Exit");

        printf("\nEnter choice: ");
        scanf("%d", &choice);

        switch (choice) {

            case 1:
                printf("Enter value: ");
                scanf("%d", &value);

                root = insert(root, value);
                break;

            case 2:
                root = deleteMin(root);
                break;

            case 3:
                printf("Enter value to search: ");
                scanf("%d", &value);

                if (search(root, value))
                    printf("Element found\n");
                else
                    printf("Element not found\n");

            case 4:

    }
